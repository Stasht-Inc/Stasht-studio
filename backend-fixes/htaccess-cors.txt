# Add this to your Laravel backend's public/.htaccess file
# Above the existing Laravel rewrite rules

# CORS Headers
<IfModule mod_headers.c>
    # Set allowed origins
    SetEnvIf Origin "^https://stashtpro\.wd-projects\.site$" AccessControlAllowOrigin=$0
    SetEnvIf Origin "^http://localhost:3000$" AccessControlAllowOrigin=$0
    SetEnvIf Origin "^http://localhost:5173$" AccessControlAllowOrigin=$0
    
    Header always set Access-Control-Allow-Origin %{AccessControlAllowOrigin}e env=AccessControlAllowOrigin
    Header always set Access-Control-Allow-Methods "GET, POST, PUT, PATCH, DELETE, OPTIONS"
    Header always set Access-Control-Allow-Headers "Origin, X-Requested-With, Content-Type, Accept, Authorization"
    Header always set Access-Control-Allow-Credentials "true"
    Header always set Access-Control-Max-Age "86400"
</IfModule>

# Handle preflight OPTIONS requests
RewriteEngine On
RewriteCond %{REQUEST_METHOD} OPTIONS
RewriteRule ^(.*)$ $1 [R=200,L]
